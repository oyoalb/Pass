<script>
let courses = JSON.parse(localStorage.getItem('pass_courses_oihan')) || [];
document.getElementById('course-date').valueAsDate = new Date();
document.getElementById('course-form').addEventListener('submit', (e) => {
e.preventDefault();
const title = document.getElementById('course-title').value;
const ue = document.getElementById('course-ue').value;
const j0Date = document.getElementById('course-date').value;
courses.push({
id: Date.now(),
title,
ue,
j0Date,
jDone: { J1: false, J3: false, J7: false, J14: false, J30: false }
});
saveData();
e.target.reset();
document.getElementById('course-date').valueAsDate = new Date();
});
function saveData() {
localStorage.setItem('pass_courses_oihan', JSON.stringify(courses));
renderTable();
}
function toggleJ(courseId, jKey) {
const course = courses.find(c => c.id === courseId);
if (course) {
course.jDone[jKey] = !course.jDone[jKey];
saveData();
}
}
function deleteCourse(id) {
if (confirm('Supprimer ce cours ?')) {
courses = courses.filter(c => c.id !== id);
saveData();
}
}
function renderTable() {
const tbody = document.getElementById('course-table-body');
const search = document.getElementById('search-input').value.toLowerCase();
const filterUe = document.getElementById('filter-ue').value;
const todayStr = new Date().toISOString().split('T');
tbody.innerHTML = '';
let totalCount = 0, todayCount = 0, completedJCount = 0;
courses.forEach(c => {
if (filterUe !== 'ALL' && c.ue !== filterUe) return;
if (search && !c.title.toLowerCase().includes(search)) return;
totalCount++;
const jIntervals = [
{ key: 'J1', days: 1 }, { key: 'J3', days: 3 },
{ key: 'J7', days: 7 }, { key: 'J14', days: 14 }, { key: 'J30', days: 30 }
];
let badgesHTML = '';
jIntervals.forEach(j => {
const d = new Date(c.j0Date);
d.setDate(d.getDate() + j.days);
const jDate = d.toISOString().split('T');
const isDone = c.jDone[j.key];
if (isDone) completedJCount++;
if (!isDone && jDate === todayStr) todayCount++;
badgesHTML += `<span class="j-badge ${isDone ? 'done' : ''}" onclick="toggleJ(${c.id}, '${j.key}')">${j.key}</span>`;
});
const tr = document.createElement('tr');
tr.innerHTML = `
<td><span class="badge-ue">${c.ue}</span></td>
<td><strong>${c.title}</strong></td>
<td>${c.j0Date}</td>
<td>${badgesHTML}</td>
<td><button class="btn-delete" onclick="deleteCourse(${c.id})">🗑️</button></td>
`;
tbody.appendChild(tr);
});
document.getElementById('stat-total').innerText = totalCount;
document.getElementById('stat-today').innerText = todayCount;
document.getElementById('stat-completed').innerText = completedJCount;
}
renderTable();
</script>
